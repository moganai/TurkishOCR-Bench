$$
\operatorname{der}[P(x)] = m
$$

$$
\operatorname{der}[Q(x)] = n
$$

$$
m > n
$$

$$
\operatorname{der}[P(x) \cdot Q(x)] = \operatorname{der}(x^m \cdot x^n) = m+n
$$

$$
\operatorname{der}[P^2(x) \cdot Q(x)] = \operatorname{der}(x^{2m} \cdot x^n) = 2m+n
$$

$$
\operatorname{der}[P(x) \pm Q(x)] = \operatorname{der}(x^m \pm x^n) = m \ (m>n)
$$

$$
\operatorname{der}\left[\frac{P(x)}{Q(x)}\right] = \operatorname{der}\left(\frac{x^m}{x^n}\right) = m-n
$$
