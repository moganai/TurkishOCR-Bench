 $$ \begin{array}{lll}
\text{der} [P(x)] = m \\
\text{der} [Q(x)] = n
\end{array}
\quad m > n $$ 

 $$ \begin{aligned}
\text{der} \left[ \rho(x). Q(x) \right] &= \text{der} (x^m x^n) = m+n \\
\text{der} \left[ \rho^2(x). Q(x) \right] &= \text{der} (x^2m x^n) = 2m+n \\
\text{der} \left[ \rho(x) \mp Q(x) \right] &= \text{der} (x^m \pm x^n = \underline{m} (m+n) \\
\text{der} \left[ \frac{\rho(x)}{\overline{Q}(x)} \right] &= \text{der} (\frac{x^m}{x^n}) = m-n
\end{aligned} $$
