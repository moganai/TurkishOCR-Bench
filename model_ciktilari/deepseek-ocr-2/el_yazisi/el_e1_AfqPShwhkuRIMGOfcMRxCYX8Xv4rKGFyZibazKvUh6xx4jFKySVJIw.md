\[ \text{der}[P(x)] = m \]  
\[ \text{der}[Q(x)] = n \]  
\[ m > n \]  

\[ \text{der}\left[\frac{P(x) \cdot Q(x)}{x^n}\right] = \frac{\text{der}\left(x^n\right) \cdot x^m}{x^{n+m}} = \frac{m+n}{x^{m+n}} \]  

\[ \text{der}\left[\frac{P(x) \cdot Q(x)}{x^2}\right] = \frac{\text{der}\left(x^2\right) \cdot x^m}{x^{m+2}} = \frac{2m+n}{x^{m+n}} \]  

\[ \text{der}\left[\frac{P(x) + Q(x)}{x^2}\right] = \frac{\text{der}\left(x^2\right) + x^m}{x^{m+2}} = \frac{m+n}{x^{m+n}} \]  

\[ \text{der}\left[\frac{P(x)}{Q(x)}\right] = \frac{\text{der}\left(x^n\right)}{Q(x)^{n+1}} = \frac{m-n}{x^{m+n}} \]
