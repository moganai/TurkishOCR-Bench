# SÖZCÜK TÜRLERİ

# İsim (ad)
- kökü -> isim
- görevi -> etkilenmek
- örnek
oda, çanta, kalem, Alya

# Sıfat
- kökü -> isim
- görevi -> ismi nitelemek, belirtmek
- ismin önüne gelir
- örnek
güzel, kirkin, kısa, uzun, besinci

# Zamir
- kökü -> isim
- görevi -> ismin yerini tutar
- örnek
ben, benim, o, onlar, şu, bu

# Zarf
- kökü -> isim
- görevi -> fiil, fiilimsi, sıfat ettilemek
- örnek
güzel, büyük, hızlı, neden, niye

# Edat
- kökü -> isim
- görevi -> kelimeler arası ilgi kurar
- örnek
ile, gibi, -e karşı, -den ötürü

# Bağlaç
- kökü -> isim
- görevi -> bağlamak
- örnek
ve, ile, veya, lakin, çünkü, de

# Ünlem
- görevi -> duygu yansıtmak
- örnek
hişt, haydi, vah vah, of, hay

# Fiil
- kökü -> fiil
- görevi -> eylemleri belirtmek
- örnek
gelmek, gitmek, almak, atfetmek
