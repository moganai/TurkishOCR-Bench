$\text{der}[P(x)] = m$

$\text{der}[Q(x)] = n$

$m > n$

$\text{der}[P(x) \cdot Q(x)] = \text{der}(x^m \cdot x^n) = m+n$

$\text{der}[P^2(x) \cdot Q(x)] = \text{der}(x^{2m} \cdot x^n) = 2m+n$

$\text{der}[P(x) \pm Q(x)] = \text{der}(x^m \pm x^n) = m \ (m>n)$

$\text{der}[\frac{P(x)}{Q(x)}] = \text{der}(\frac{x^m}{x^n}) = m-n$
